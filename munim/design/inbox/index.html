# backend/sources/paytm-pg.yaml  (proposed)
id: paytm-pg
name: Paytm Payment Gateway (public API)
kind: api
enabled: true
holds: Status of a customer payment or refund, from
  Paytm's public Transaction Status and Refund Status APIs.
when:
  topics: [settlement]
  words: [paid, order, refund, "not received"]
api:
  preset: paytm-pg      # new: signed POST (checksum with the key)
  mode: recorded       # recorded | staging
  mid: ${PAYTM_MID}
  key: ${PAYTM_KEY_SECRET}  # in .env, never in this file
